    85→// --- MTN MoMo : intégration officielle (Request to Pay) ---
    86→const mtnMomoProvider: PaymentProviderInterface = {
    87→  code: "MTN_MOMO",
    88→  name: "MTN Mobile Money",
    89→  isConfigured() {
    90→    return momoCollectionConfigured();
    91→  },
    92→  async initializePayment(ctx) {
    93→    if (!this.isConfigured()) {
    94→      throw new ApiError(
    95→        503,
    96→        "MOMO_NOT_CONFIGURED",
    97→        "Paiement MTN MoMo non configuré. Renseignez les clés MOMO_COLLECTION_* (voir README)."
    98→      );
    99→    }
   100→    // ---- Request to Pay réel ----
   101→    if (!ctx.phone) {
   102→      throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   103→    }
   104→    const msisdn = toMomoMsisdn(ctx.phone); // 400 si invalide
   105→    const referenceId = randomUUID(); // X-Reference-Id — idempotence MoMo
   106→
   107→    await momoRequestToPay({
   108→      referenceId,
   109→      amount: ctx.amount,
   110→      currency: momoCurrency(),
   111→      externalId: ctx.bookingReference ?? ctx.paymentId,
   112→      msisdn,
   113→      payerMessage: ctx.bookingReference
   114→        ? `Billet ${ctx.bookingReference} — NZOKO TRANSPORT`
   115→        : "NZOKO TRANSPORT",
   116→      payeeNote: `Paiement billet ${ctx.bookingReference ?? ""}`.trim(),
   117→    });
   118→
   119→    return {
   120→      status: "PROCESSING",
   121→      providerTransactionId: referenceId,
   122→      instructions:
   123→        "Demande de paiement envoyée sur votre téléphone. Validez-la depuis votre application MTN MoMo ou le menu MoMo de votre SIM, puis patientez quelques secondes.",
   124→    };
   125→  },
   126→  async verifyPayment(providerTransactionId) {
   127→    if (!this.isConfigured()) return "PROCESSING";
   128→    const st = await momoGetRequestToPayStatus(providerTransactionId);
   129→    return st.status === "SUCCESSFUL" ? "SUCCESS" : st.status === "FAILED" ? "FAILED" : "PROCESSING";
   130→  },
   131→  async refundPayment(providerTransactionId) {
   132→    if (!momoDisbursementConfigured()) return false;
   133→    await momoRefund({
   134→      referenceId: randomUUID(),
   135→      amount: 0, // remplacé par le contexte réel dans initiatePaymentRefund
   136→      currency: momoCurrency(),
   137→      externalId: providerTransactionId,
   138→      referenceIdToRefund: providerTransactionId,
   139→    });
   140→    return true; // initiation acceptée — suivi par pollPaymentRefundStatus
   141→  },
   142→};

   242→async function writeMetadata(paymentId: string, mutate: (meta: PaymentMetadata) => PaymentMetadata): Promise<void> {
   243→  const payment = await db.payment.findUnique({ where: { id: paymentId }, select: { metadata: true } });
   244→  if (!payment) return;
   245→  const next = mutate(parseMetadata(payment.metadata));
   246→  await db.payment.update({ where: { id: paymentId }, data: { metadata: JSON.stringify(next) } });
   247→}

   253→export async function createPayment(
   254→  input: PayBookingInput,
   255→  ctx: { actorUserId?: string | null; ip?: string | null }
   256→): Promise<PaymentDTO> {
   257→  const booking = await db.booking.findUnique({
   258→    where: { id: input.bookingId },
   259→    include: { trip: true, payment: true },
   260→  });
   261→  if (!booking) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Réservation introuvable.");
   262→  if (booking.status !== "PENDING") {
   263→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette réservation n'est plus en attente de paiement.");
   264→  }
   265→  if (booking.expiresAt && booking.expiresAt < new Date()) {
   266→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le délai de réservation a expiré. Veuillez réserver à nouveau.");
   267→  }
   268→  if (booking.payment.some((p) => ["PENDING", "PROCESSING", "SUCCESS"].includes(p.status))) {
   269→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un paiement est déjà en cours pour cette réservation.");
   270→  }
   271→
   272→  const provider = getProvider(input.provider);
   273→
   274→  // MTN MoMo réel : validation E.164 AVANT toute écriture (400 propre, aucune ligne créée)
   275→  let momoPhone = input.momoPhone;
   276→  if (input.provider === "MTN_MOMO" && provider.isConfigured()) {
   277→    if (!momoPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   278→    momoPhone = toMomoMsisdn(momoPhone);
   279→  }
   280→
   281→  const payment = await db.payment.create({
   282→    data: {
   283→      bookingId: booking.id,
   284→      provider: input.provider,
   285→      amount: booking.amount,
   286→      status: "PENDING",
   287→      createdById: ctx.actorUserId ?? null,
   288→    },
   289→  });
   290→
   291→  let result: ProviderResult;
   292→  try {
   293→    result = await provider.initializePayment({
   294→      paymentId: payment.id,
   295→      amount: booking.amount,
   296→      phone: momoPhone,
   297→      senderName: input.senderName,
   298→      bookingReference: booking.bookingReference,
   299→    });
   300→  } catch (err) {
   301→    // Échec d'initialisation fournisseur (ex : numéro MoMo introuvable) →
   302→    // le paiement est marqué FAILED (réessayable) et l'erreur remonte au client.
   303→    const reason = err instanceof ApiError ? err.message : "Initialisation du paiement impossible.";
   304→    await db.payment.update({
   305→      where: { id: payment.id },
   306→      data: { status: "FAILED", metadata: JSON.stringify({ reason }) },
   307→    });
   308→    throw err;
   309→  }
   310→
   311→  const momoMeta: PaymentMetadata["momo"] | undefined =
   312→    input.provider === "MTN_MOMO" && result.providerTransactionId
   313→      ? {
   314→          product: "collection",
   315→          referenceId: result.providerTransactionId,
   316→          msisdn: momoPhone ?? "",
   317→          currency: momoCurrency(),
   318→          environment: momoConfigStatus().environment,
   319→        }
   320→      : undefined;
   321→
   322→  const updated = await db.payment.update({
   323→    where: { id: payment.id },
   324→    data: {
   325→      status: result.status,
   326→      providerTransactionId: result.providerTransactionId,
   327→      metadata: JSON.stringify({
   328→        instructions: result.instructions,
   329→        momoPhone: momoPhone ?? null,
   330→        senderName: input.senderName ?? null,
   331→        ip: ctx.ip ?? null,
   332→        ...(momoMeta ? { momo: momoMeta } : {}),
   333→      }),
   334→    },
   335→    include: { createdBy: true, booking: true },
   336→  });
   337→
   338→  await logAudit({
   339→    userId: ctx.actorUserId ?? null,
   340→    action: "PAYMENT_INITIALIZED",
   341→    entity: "Payment",
   342→    entityId: payment.id,
   343→    metadata: {
   344→      provider: input.provider,
   345→      amount: booking.amount,
   346→      ...(result.providerTransactionId ? { momoReferenceId: result.providerTransactionId } : {}),
   347→    },
   348→    ipAddress: ctx.ip ?? null,
   349→  });
   350→
   351→  return toPaymentDTO(updated);
   352→}

   354→// ------------------------------------------------------------
   355→// Confirmation idempotente + émission billet
   356→// Cœur du système : appelée par cash, webhook, confirmation manuelle.

   358→
   359→export async function confirmPaymentAndIssueTicket(
   360→  paymentId: string,
   361→  actorUserId: string | null,
   362→  ip: string | null
   363→): Promise<{ payment: PaymentDTO; alreadyConfirmed: boolean }> {
   364→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   365→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   366→
   367→  if (payment.status === "SUCCESS") {
   368→    return { payment: toPaymentDTO(payment), alreadyConfirmed: true }; // idempotence stricte
   369→  }
   370→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   371→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Ce paiement ne peut plus être confirmé.");
   372→  }
   373→
   374→  await db.$transaction(async (tx) => {
   375→    // Re-vérification atomique du statut dans la transaction
   376→    const fresh = await tx.payment.findUnique({ where: { id: payment.id }, select: { status: true } });
   377→    if (!fresh || fresh.status !== payment.status) {
   378→      throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le paiement a changé d'état concurrentiellement.");
   379→    }
   380→
   381→    await tx.payment.update({ where: { id: payment.id }, data: { status: "SUCCESS" } });
   382→    await tx.booking.update({ where: { id: payment.bookingId }, data: { status: "CONFIRMED" } });
   383→    await tx.seatOccupancy.updateMany({
   384→      where: { bookingId: payment.bookingId },
   385→      data: { status: "BOOKED", expiresAt: null },
   386→    });
   387→
   388→    // Billet émis exactement une fois (contrainte unique bookingId)
   389→    await issueTicketForBooking(tx, payment.bookingId, payment.booking.bookingReference);
   390→
   391→    // Écriture comptable idempotente (référence unique)
   392→    await tx.transaction.create({
   393→      data: {
   394→        type: "INCOME",
   395→        amount: payment.amount,
   396→        reference: payment.booking.bookingReference,
   397→        description: `Vente billet ${payment.booking.bookingReference} (${payment.provider})`,
   398→        agencyId: payment.booking.agencyId,
   399→        paymentId: payment.id,
   400→        createdById: actorUserId,
   401→      },
   402→    });
   403→  });
   404→
   405→  await logAudit({
   406→    userId: actorUserId,
   407→    action: "PAYMENT_CONFIRMED",
   408→    entity: "Payment",
   409→    entityId: payment.id,
   410→    metadata: { provider: payment.provider, amount: payment.amount, booking: payment.booking.bookingReference },
   411→    ipAddress: ip,
   412→  });
   413→
   414→  const final = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   415→  return { payment: toPaymentDTO(final!), alreadyConfirmed: false };
   416→}

   418→// Confirmation espèces par un agent (permission payment:cash-collect)
   419→export async function confirmCashPayment(paymentId: string, actorUserId: string, ip: string | null): Promise<PaymentDTO> {
   420→  const payment = await db.payment.findUnique({ where: { id: paymentId } });
   421→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   422→  if (payment.provider !== "CASH") {
   423→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette confirmation ne concerne qu'un paiement en espèces.");
   424→  }
   425→  const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, actorUserId, ip);
   426→  return dto;
   427→}

   430→// MTN MoMo — suivi Request to Pay (polling public, idempotent)

   433→
   434→export async function pollMomoPaymentStatus(
   435→  paymentId: string,
   436→  ctx: { actorUserId?: string | null; ip?: string | null }
   437→): Promise<PaymentDTO> {
   438→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   439→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   440→  if (payment.provider !== "MTN_MOMO") {
   441→    throw new ApiError(400, ERROR_CODES.BAD_REQUEST, "Ce paiement n'est pas un paiement MTN Mobile Money.");
   442→  }
   443→  // Statut déjà final → réponse idempotente (pas d'appel MTN inutile)
   444→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   445→    return toPaymentDTO(payment);
   446→  }
   447→  if (!payment.providerTransactionId) {
   448→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Référence de transaction MoMo introuvable pour ce paiement.");
   449→  }
   450→
   451→  const status = await momoGetRequestToPayStatus(payment.providerTransactionId);
   452→
   453→  if (status.status === "SUCCESSFUL") {
   454→    // Confirmation idempotente (billet + écriture comptable) — le cœur existant
   455→    const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, ctx.actorUserId ?? null, ctx.ip ?? null);
   456→    await writeMetadata(paymentId, (m) => ({
   457→      ...m,
   458→      momo: m.momo
   459→        ? { ...m.momo, financialTransactionId: status.financialTransactionId }
   460→        : m.momo,
   461→    }));
   462→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   463→    return toPaymentDTO(fresh!);
   464→  }
   465→
   466→  if (status.status === "FAILED") {
   467→    await db.payment.update({ where: { id: paymentId }, data: { status: "FAILED" } });
   468→    await writeMetadata(paymentId, (m) => ({
   469→      ...m,
   470→      reason: momoFailureMessage(status),
   471→      momo: m.momo ? { ...m.momo, reasonCode: status.reasonCode, reason: status.reasonMessage } : m.momo,
   472→    }));
   473→    await logAudit({
   474→      userId: ctx.actorUserId ?? null,
   475→      action: "MOMO_PAYMENT_FAILED",
   476→      entity: "Payment",
   477→      entityId: paymentId,
   478→      metadata: { reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   479→      ipAddress: ctx.ip ?? null,
   480→    });
   481→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   482→    return toPaymentDTO(fresh!);
   483→  }
   484→
   485→  // Toujours PENDING côté MTN
   486→  return toPaymentDTO(payment);
   487→}

   489→// ------------------------------------------------------------
   490→// MTN MoMo — REMBOURSEMENTS (envoi de fonds aux clients)
   491→// Modes : MOMO_REFUND (Refund API, référence R2P d'origine),
   492→//         MOMO_TRANSFER (Transfer vers un MSISDN),
   493→//         CASH (espèces rendues au guichet — finalisation immédiate).
   494→// ------------------------------------------------------------
   495→
   496→async function finalizeRefund(
   497→  paymentId: string,
   498→  mode: RefundPaymentInput["mode"],
   499→  refundMeta: NonNullable<PaymentMetadata["refund"]>,
   500→  actorUserId: string | null,
   501→  ip: string | null,
   502→  financialTransactionId: string | null
   503→): Promise<void> {
   504→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true } });
   505→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   506→
   507→  const confirmedMeta: NonNullable<PaymentMetadata["refund"]> = {
   508→    ...refundMeta,
   509→    status: "SUCCESSFUL",
   510→    confirmedAt: new Date().toISOString(),
   511→    financialTransactionId,
   512→  };
   513→
   514→  await db.$transaction(async (tx) => {
   515→    // Statut + métadonnées de remboursement de façon ATOMIQUE (même transaction)
   516→    await tx.payment.update({
   517→      where: { id: paymentId },
   518→      data: {
   519→        status: "REFUNDED",
   520→        metadata: JSON.stringify({ ...parseMetadata(payment.metadata), refund: confirmedMeta }),
   521→      },
   522→    });
   523→    // Écriture comptable REFUND (aucun doublon possible : le statut REFUNDED
   524→    // fait échouer toute nouvelle tentative de remboursement ensuite)
   525→    await tx.transaction.create({
   526→      data: {
   527→        type: "REFUND",
   528→        amount: refundMeta.amount,
   529→        reference: payment.booking.bookingReference,
   530→        description: `Remboursement billet ${payment.booking.bookingReference} (${
   531→          mode === "CASH" ? "espèces" : mode === "MOMO_REFUND" ? "MTN MoMo — remboursement" : "MTN MoMo — transfert"
   532→        }${refundMeta.msisdn ? ` vers ${refundMeta.msisdn}` : ""})`,
   533→        agencyId: payment.booking.agencyId,
   534→        paymentId: paymentId,
   535→        createdById: actorUserId,
   536→      },
   537→    });
   538→  });
   539→
   540→  await logAudit({
   541→    userId: actorUserId,
   542→    action: "REFUND_CONFIRMED",
   543→    entity: "Payment",
   544→    entityId: paymentId,
   545→    metadata: { mode, amount: refundMeta.amount, booking: payment.booking.bookingReference, financialTransactionId },
   546→    ipAddress: ip,
   547→  });
   548→}

   550→export async function initiatePaymentRefund(
   551→  paymentId: string,
   552→  input: RefundPaymentInput,
   553→  ctx: { userId: string; userName: string; ip?: string | null }
   554→): Promise<PaymentDTO> {
   555→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   556→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   557→  if (payment.status !== "SUCCESS") {
   558→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Seul un paiement réussi peut être remboursé.");
   559→  }
   560→  const meta = parseMetadata(payment.metadata);
   561→  if (meta.refund?.status === "PROCESSING" || meta.refund?.status === "SUCCESSFUL") {
   562→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un remboursement est déjà en cours ou terminé pour ce paiement.");
   563→  }
   564→  if (input.amount !== undefined && (input.amount <= 0 || input.amount > payment.amount)) {
   565→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Montant de remboursement invalide.");
   566→  }
   567→  const refundAmount = input.amount ?? payment.amount;
   568→  const bookingRef = payment.booking.bookingReference;
   569→
   570→  // ---- Espèces : finalisation immédiate ----
   571→  if (input.mode === "CASH") {
   572→    const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   573→      mode: "CASH",
   574→      status: "PROCESSING",
   575→      amount: refundAmount,
   576→      initiatedByName: ctx.userName,
   577→      initiatedAt: new Date().toISOString(),
   578→    };
   579→    await finalizeRefund(paymentId, "CASH", refundMeta, ctx.userId, ctx.ip ?? null, null);
   580→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   581→    return toPaymentDTO(fresh!);
   582→  }
   583→
   584→  // ---- Modes MoMo : disbursement obligatoire ----
   585→  if (!momoDisbursementConfigured()) {
   586→    throw new ApiError(
   587→      503,
   588→      "MOMO_NOT_CONFIGURED",
   589→      "Remboursement MTN MoMo non configuré. Renseignez les clés MOMO_DISBURSEMENT_* (voir README)."
   590→    );
   591→  }
   592→  const referenceId = randomUUID();
   593→  let msisdn: string | null = null;
   594→
   595→  if (input.mode === "MOMO_REFUND") {
   596→    if (payment.provider !== "MTN_MOMO" || !payment.providerTransactionId) {
   597→      throw new ApiError(
   598→        409,
   599→        ERROR_CODES.PAYMENT_ERROR,
   600→        "Le remboursement automatique MTN nécessite un paiement MTN MoMo (transaction d'origine requise)."
   601→      );
   602→    }
   603→    await momoRefund({
   604→      referenceId,
   605→      amount: refundAmount,
   606→      currency: momoCurrency(),
   607→      externalId: bookingRef,
   608→      referenceIdToRefund: payment.providerTransactionId,
   609→      payerMessage: `Remboursement billet ${bookingRef}`,
   610→      payeeNote: "Remboursement NZOKO TRANSPORT",
   611→    });
   612→  } else {
   613→    // MOMO_TRANSFER — envoi de fonds vers le MSISDN du passager (ou saisi)
   614→    const rawPhone: string | null = input.msisdn ?? meta.momoPhone ?? null;
   615→    if (!rawPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money du bénéficiaire requis.");
   616→    msisdn = toMomoMsisdn(rawPhone);
   617→    await momoTransfer({
   618→      referenceId,
   619→      amount: refundAmount,
   620→      currency: momoCurrency(),
   621→      externalId: bookingRef,
   622→      msisdn,
   623→      payerMessage: `Remboursement billet ${bookingRef}`,
   624→      payeeNote: "Remboursement NZOKO TRANSPORT",
   625→    });
   626→  }
   627→
   628→  const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   629→    mode: input.mode,
   630→    status: "PROCESSING",
   631→    amount: refundAmount,
   632→    referenceId,
   633→    msisdn,
   634→    initiatedByName: ctx.userName,
   635→    initiatedAt: new Date().toISOString(),
   636→  };
   637→  await writeMetadata(paymentId, (m) => ({ ...m, refund: refundMeta }));
   638→
   639→  await logAudit({
   640→    userId: ctx.userId,
   641→    action: "REFUND_INITIATED",
   642→    entity: "Payment",
   643→    entityId: paymentId,
   644→    metadata: { mode: input.mode, amount: refundAmount, booking: bookingRef, momoReferenceId: referenceId },
   645→    ipAddress: ctx.ip ?? null,
   646→  });
   647→
   648→  const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   649→  return toPaymentDTO(fresh!);
   650→}

   652→export async function pollPaymentRefundStatus(
   653→  paymentId: string,
   654→  ctx: { userId: string; ip?: string | null }
   655→): Promise<PaymentDTO> {
   656→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   657→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   658→  const meta = parseMetadata(payment.metadata);
   659→  const refund = meta.refund;
   660→  if (!refund) throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Aucun remboursement initié pour ce paiement.");
   661→  if (refund.status !== "PROCESSING" || !refund.referenceId) {
   662→    return toPaymentDTO(payment); // déjà final (SUCCESSFUL/FAILED) → idempotent
   663→  }
   664→
   665→  const status =
   666→    refund.mode === "MOMO_REFUND"
   667→      ? await momoGetRefundStatus(refund.referenceId)
   668→      : await momoGetTransferStatus(refund.referenceId);
   669→
   670→  if (status.status === "SUCCESSFUL") {
   671→    await finalizeRefund(paymentId, refund.mode, refund, ctx.userId, ctx.ip ?? null, status.financialTransactionId);
   672→  } else if (status.status === "FAILED") {
   673→    await writeMetadata(paymentId, (m) => ({
   674→      ...m,
   675→      refund: m.refund ? { ...m.refund, status: "FAILED", reason: momoFailureMessage(status), reasonCode: status.reasonCode } : m.refund,
   676→    }));
   677→    await logAudit({
   678→      userId: ctx.userId,
   679→      action: "REFUND_FAILED",
   680→      entity: "Payment",
   681→      entityId: paymentId,
   682→      metadata: { mode: refund.mode, reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   683→      ipAddress: ctx.ip ?? null,
   684→    });
   685→  }
   686→
   687→  const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   688→  return toPaymentDTO(fresh!);
   689→}

   733→export async function handleMomoWebhook(rawBody: string, ip: string | null): Promise<{ received: true; status: string }> {
   734→  const payload = safeJsonParse<{ referenceId?: string; status?: string } | null>(rawBody, null);
   735→  if (!payload?.referenceId) {
   736→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "momo callback sans referenceId" } });
   737→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload callback MoMo invalide.");
   738→  }
   739→
   740→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.referenceId } });
   741→  if (!payment) {
   742→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction MoMo inconnue.");
   743→  }
   744→
   745→  // Re-vérification systématique auprès de MTN (jamais sur parole du callback)
   746→  const dto = await pollMomoPaymentStatus(payment.id, { ip });
   747→  return { received: true, status: dto.status };
   748→}

   750→// ------------------------------------------------------------
   751→// Webhook fournisseur interne — HMAC, idempotent, jamais sur parole du client
   752→// POST /api/webhooks/payments
   753→// Signature attendue : X-Nzoko-Signature: hex HMAC-SHA256(body, WEBHOOK_SECRET)
   754→// ------------------------------------------------------------
   755→export interface WebhookPayload {
   756→  provider: string;
   757→  providerTransactionId: string;
   758→  status: "SUCCESS" | "FAILED" | "CANCELLED";
   759→  amount?: number;
   760→}
   761→
   762→export async function handlePaymentWebhook(rawBody: string, signature: string | null, ip: string | null): Promise<{ received: true; duplicate: boolean }> {
   763→  const secret = process.env.WEBHOOK_SECRET;
   764→  if (!secret) throw new ApiError(503, "WEBHOOK_NOT_CONFIGURED", "Webhook non configuré.");
   765→
   766→  if (!signature || !timingSafeEqual(signature, hmacSha256(secret, rawBody))) {
   767→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "signature invalide" } });
   768→    throw new ApiError(401, ERROR_CODES.UNAUTHORIZED, "Signature webhook invalide.");
   769→  }
   770→
   771→  const payload = safeJsonParse<WebhookPayload | null>(rawBody, null);
   772→  if (!payload?.providerTransactionId || !payload.status) {
   773→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload webhook invalide.");
   774→  }
   775→
   776→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.providerTransactionId } });
   777→  if (!payment) {
   778→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "transaction inconnue", tx: payload.providerTransactionId } });
   779→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction inconnue.");
   780→  }
   781→
   782→  if (payload.amount !== undefined && payload.amount !== payment.amount) {
   783→    await logSecurity({ event: "SUSPICIOUS", ipAddress: ip, details: { reason: "montant webhook divergent", paymentId: payment.id } });
   784→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Montant divergent — paiement bloqué.");
   785→  }
   786→
   787→  if (payload.status !== "SUCCESS") {
   788→    await db.payment.update({ where: { id: payment.id }, data: { status: payload.status } });
   789→    return { received: true, duplicate: false };
   790→  }
   791→
   792→  // Idempotence : re-confirmer un paiement déjà SUCCESS est un no-op réussi
   793→  const { alreadyConfirmed } = await confirmPaymentAndIssueTicket(payment.id, null, ip);
   794→  return { received: true, duplicate: alreadyConfirmed };
   795→}
