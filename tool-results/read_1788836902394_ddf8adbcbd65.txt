    84→// --- MTN MoMo : intégration officielle (Request to Pay) ---
    85→const mtnMomoProvider: PaymentProviderInterface = {
    86→  code: "MTN_MOMO",
    87→  name: "MTN Mobile Money",
    88→  isConfigured() {
    89→    return momoCollectionConfigured();
    90→  },
    91→  async initializePayment(ctx) {
    92→    if (!this.isConfigured()) {
    93→      throw new ApiError(
    94→        503,
    95→        "MOMO_NOT_CONFIGURED",
    96→        "Paiement MTN MoMo non configuré. Renseignez les clés MOMO_COLLECTION_* (voir README)."
    97→      );
    98→    }
    99→    // ---- Request to Pay réel ----
   100→    if (!ctx.phone) {
   101→      throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   102→    }
   103→    const msisdn = toMomoMsisdn(ctx.phone); // 400 si invalide
   104→    const referenceId = randomUUID(); // X-Reference-Id — idempotence MoMo
   105→
   106→    await momoRequestToPay({
   107→      referenceId,
   108→      amount: ctx.amount,
   109→      currency: momoCurrency(),
   110→      externalId: ctx.bookingReference ?? ctx.paymentId,
   111→      msisdn,
   112→      payerMessage: ctx.bookingReference
   113→        ? `Billet ${ctx.bookingReference} — NZOKO TRANSPORT`
   114→        : "NZOKO TRANSPORT",
   115→      payeeNote: `Paiement billet ${ctx.bookingReference ?? ""}`.trim(),
   116→    });
   117→
   118→    return {
   119→      status: "PROCESSING",
   120→      providerTransactionId: referenceId,
   121→      instructions:
   122→        "Demande de paiement envoyée sur votre téléphone. Validez-la depuis votre application MTN MoMo ou le menu MoMo de votre SIM, puis patientez quelques secondes.",
   123→    };
   124→  },
   125→  async verifyPayment(providerTransactionId) {
   126→    if (!this.isConfigured()) return "PROCESSING";
   127→    const st = await momoGetRequestToPayStatus(providerTransactionId);
   128→    return st.status === "SUCCESSFUL" ? "SUCCESS" : st.status === "FAILED" ? "FAILED" : "PROCESSING";
   129→  },
   130→  async refundPayment(providerTransactionId) {
   131→    if (!momoDisbursementConfigured()) return false;
   132→    await momoRefund({
   133→      referenceId: randomUUID(),
   134→      amount: 0, // remplacé par le contexte réel dans initiatePaymentRefund
   135→      currency: momoCurrency(),
   136→      externalId: providerTransactionId,
   137→      referenceIdToRefund: providerTransactionId,
   138→    });
   139→    return true; // initiation acceptée — suivi par pollPaymentRefundStatus
   140→  },
   141→};

   237→function parseMetadata(raw: string | null): PaymentMetadata {
   238→  return safeJsonParse<PaymentMetadata>(raw ?? "{}", {});

   241→async function writeMetadata(paymentId: string, mutate: (meta: PaymentMetadata) => PaymentMetadata): Promise<void> {
   242→  const payment = await db.payment.findUnique({ where: { id: paymentId }, select: { metadata: true } });
   243→  if (!payment) return;
   244→  const next = mutate(parseMetadata(payment.metadata));
   245→  await db.payment.update({ where: { id: paymentId }, data: { metadata: JSON.stringify(next) } });
   246→}
   247→
   248→// Relit un paiement avec ses relations après mutation — lève 404 plutôt
   249→// qu'une assertion non-null (le paiement vient d'être écrit : absence =
   250→// suppression concurrente, signalée proprement à l'appelant).
   251→async function refetchPayment(paymentId: string) {
   252→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   253→  if (!payment) {
   254→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   255→  }
   256→  return payment;
   257→}

   263→export async function createPayment(
   264→  input: PayBookingInput,
   265→  ctx: { actorUserId?: string | null; ip?: string | null }
   266→): Promise<PaymentDTO> {
   267→  const booking = await db.booking.findUnique({
   268→    where: { id: input.bookingId },
   269→    include: { trip: true, payment: true },
   270→  });
   271→  if (!booking) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Réservation introuvable.");
   272→  if (booking.status !== "PENDING") {
   273→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette réservation n'est plus en attente de paiement.");
   274→  }
   275→  if (booking.expiresAt && booking.expiresAt < new Date()) {
   276→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le délai de réservation a expiré. Veuillez réserver à nouveau.");
   277→  }
   278→  if (booking.payment.some((p) => ["PENDING", "PROCESSING", "SUCCESS"].includes(p.status))) {
   279→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un paiement est déjà en cours pour cette réservation.");
   280→  }
   281→
   282→  const provider = getProvider(input.provider);
   283→
   284→  // MTN MoMo réel : validation E.164 AVANT toute écriture (400 propre, aucune ligne créée)
   285→  let momoPhone = input.momoPhone;
   286→  if (input.provider === "MTN_MOMO" && provider.isConfigured()) {
   287→    if (!momoPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   288→    momoPhone = toMomoMsisdn(momoPhone);
   289→  }
   290→
   291→  const payment = await db.payment.create({
   292→    data: {
   293→      bookingId: booking.id,
   294→      provider: input.provider,
   295→      amount: booking.amount,
   296→      status: "PENDING",
   297→      createdById: ctx.actorUserId ?? null,
   298→    },
   299→  });
   300→
   301→  let result: ProviderResult;
   302→  try {
   303→    result = await provider.initializePayment({
   304→      paymentId: payment.id,
   305→      amount: booking.amount,
   306→      phone: momoPhone,
   307→      senderName: input.senderName,
   308→      bookingReference: booking.bookingReference,
   309→    });
   310→  } catch (err) {
   311→    // Échec d'initialisation fournisseur (ex : numéro MoMo introuvable) →
   312→    // le paiement est marqué FAILED (réessayable) et l'erreur remonte au client.
   313→    const reason = err instanceof ApiError ? err.message : "Initialisation du paiement impossible.";
   314→    await db.payment.update({
   315→      where: { id: payment.id },
   316→      data: { status: "FAILED", metadata: JSON.stringify({ reason }) },
   317→    });
   318→    throw err;
   319→  }
   320→
   321→  const momoMeta: PaymentMetadata["momo"] | undefined =
   322→    input.provider === "MTN_MOMO" && result.providerTransactionId
   323→      ? {
   324→          product: "collection",
   325→          referenceId: result.providerTransactionId,
   326→          msisdn: momoPhone ?? "",
   327→          currency: momoCurrency(),
   328→          environment: momoConfigStatus().environment,
   329→        }
   330→      : undefined;
   331→
   332→  const updated = await db.payment.update({
   333→    where: { id: payment.id },
   334→    data: {
   335→      status: result.status,
   336→      providerTransactionId: result.providerTransactionId,
   337→      metadata: JSON.stringify({
   338→        instructions: result.instructions,
   339→        momoPhone: momoPhone ?? null,
   340→        senderName: input.senderName ?? null,
   341→        ip: ctx.ip ?? null,
   342→        ...(momoMeta ? { momo: momoMeta } : {}),
   343→      }),
   344→    },
   345→    include: { createdBy: true, booking: true },
   346→  });
   347→
   348→  await logAudit({
   349→    userId: ctx.actorUserId ?? null,
   350→    action: "PAYMENT_INITIALIZED",
   351→    entity: "Payment",
   352→    entityId: payment.id,
   353→    metadata: {
   354→      provider: input.provider,
   355→      amount: booking.amount,
   356→      ...(result.providerTransactionId ? { momoReferenceId: result.providerTransactionId } : {}),
   357→    },
   358→    ipAddress: ctx.ip ?? null,
   359→  });
   360→
   361→  return toPaymentDTO(updated);
   362→}

   364→// ------------------------------------------------------------
   365→// Confirmation idempotente + émission billet
   366→// Cœur du système : appelée par cash, webhook, confirmation manuelle.
   367→// ------------------------------------------------------------
   368→
   369→export async function confirmPaymentAndIssueTicket(
   370→  paymentId: string,
   371→  actorUserId: string | null,
   372→  ip: string | null
   373→): Promise<{ payment: PaymentDTO; alreadyConfirmed: boolean }> {
   374→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   375→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   376→
   377→  if (payment.status === "SUCCESS") {
   378→    return { payment: toPaymentDTO(payment), alreadyConfirmed: true }; // idempotence stricte
   379→  }
   380→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   381→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Ce paiement ne peut plus être confirmé.");
   382→  }
   383→
   384→  await db.$transaction(async (tx) => {
   385→    // Re-vérification atomique du statut dans la transaction
   386→    const fresh = await tx.payment.findUnique({ where: { id: payment.id }, select: { status: true } });
   387→    if (!fresh || fresh.status !== payment.status) {
   388→      throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le paiement a changé d'état concurrentiellement.");
   389→    }
   390→
   391→    await tx.payment.update({ where: { id: payment.id }, data: { status: "SUCCESS" } });
   392→    await tx.booking.update({ where: { id: payment.bookingId }, data: { status: "CONFIRMED" } });
   393→    await tx.seatOccupancy.updateMany({
   394→      where: { bookingId: payment.bookingId },
   395→      data: { status: "BOOKED", expiresAt: null },
   396→    });
   397→
   398→    // Billet émis exactement une fois (contrainte unique bookingId)
   399→    await issueTicketForBooking(tx, payment.bookingId, payment.booking.bookingReference);
   400→
   401→    // Écriture comptable idempotente (référence unique)
   402→    await tx.transaction.create({
   403→      data: {
   404→        type: "INCOME",
   405→        amount: payment.amount,
   406→        reference: payment.booking.bookingReference,
   407→        description: `Vente billet ${payment.booking.bookingReference} (${payment.provider})`,
   408→        agencyId: payment.booking.agencyId,
   409→        paymentId: payment.id,
   410→        createdById: actorUserId,
   411→      },
   412→    });
   413→  });
   414→
   415→  await logAudit({
   416→    userId: actorUserId,
   417→    action: "PAYMENT_CONFIRMED",
   418→    entity: "Payment",
   419→    entityId: payment.id,
   420→    metadata: { provider: payment.provider, amount: payment.amount, booking: payment.booking.bookingReference },
   421→    ipAddress: ip,
   422→  });
   423→
   424→  const final = await refetchPayment(paymentId);
   425→  return { payment: toPaymentDTO(final), alreadyConfirmed: false };
   426→}

   428→// Confirmation espèces par un agent (permission payment:cash-collect)
   429→export async function confirmCashPayment(paymentId: string, actorUserId: string, ip: string | null): Promise<PaymentDTO> {
   430→  const payment = await db.payment.findUnique({ where: { id: paymentId } });
   431→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   432→  if (payment.provider !== "CASH") {
   433→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette confirmation ne concerne qu'un paiement en espèces.");
   434→  }
   435→  const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, actorUserId, ip);
   436→  return dto;
   437→}

   439→// ------------------------------------------------------------
   440→// MTN MoMo — suivi Request to Pay (polling public, idempotent)
   441→// Référence : GET /collection/v1_0/requesttopay/{referenceId}
   442→// ------------------------------------------------------------
   443→
   444→export async function pollMomoPaymentStatus(
   445→  paymentId: string,
   446→  ctx: { actorUserId?: string | null; ip?: string | null }
   447→): Promise<PaymentDTO> {
   448→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   449→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   450→  if (payment.provider !== "MTN_MOMO") {
   451→    throw new ApiError(400, ERROR_CODES.BAD_REQUEST, "Ce paiement n'est pas un paiement MTN Mobile Money.");
   452→  }
   453→  // Statut déjà final → réponse idempotente (pas d'appel MTN inutile)
   454→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   455→    return toPaymentDTO(payment);
   456→  }
   457→  if (!payment.providerTransactionId) {
   458→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Référence de transaction MoMo introuvable pour ce paiement.");
   459→  }
   460→
   461→  const status = await momoGetRequestToPayStatus(payment.providerTransactionId);
   462→
   463→  if (status.status === "SUCCESSFUL") {
   464→    // Confirmation idempotente (billet + écriture comptable) — le cœur existant
   465→    await confirmPaymentAndIssueTicket(paymentId, ctx.actorUserId ?? null, ctx.ip ?? null);
   466→    await writeMetadata(paymentId, (m) => ({
   467→      ...m,
   468→      momo: m.momo
   469→        ? { ...m.momo, financialTransactionId: status.financialTransactionId }
   470→        : m.momo,
   471→    }));
   472→    return toPaymentDTO(await refetchPayment(paymentId));
   473→  }
   474→
   475→  if (status.status === "FAILED") {
   476→    await db.payment.update({ where: { id: paymentId }, data: { status: "FAILED" } });
   477→    await writeMetadata(paymentId, (m) => ({
   478→      ...m,
   479→      reason: momoFailureMessage(status),
   480→      momo: m.momo ? { ...m.momo, reasonCode: status.reasonCode, reason: status.reasonMessage } : m.momo,
   481→    }));
   482→    await logAudit({
   483→      userId: ctx.actorUserId ?? null,
   484→      action: "MOMO_PAYMENT_FAILED",
   485→      entity: "Payment",
   486→      entityId: paymentId,
   487→      metadata: { reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   488→      ipAddress: ctx.ip ?? null,
   489→    });
   490→    return toPaymentDTO(await refetchPayment(paymentId));
   491→  }
   492→
   493→  // Toujours PENDING côté MTN
   494→  return toPaymentDTO(payment);
   495→}

   497→// ------------------------------------------------------------
   498→// MTN MoMo — REMBOURSEMENTS (envoi de fonds aux clients)
   499→// Modes : MOMO_REFUND (Refund API, référence R2P d'origine),
   500→//         MOMO_TRANSFER (Transfer vers un MSISDN),
   501→//         CASH (espèces rendues au guichet — finalisation immédiate).
   502→// ------------------------------------------------------------
   503→
   504→async function finalizeRefund(
   505→  paymentId: string,
   506→  mode: RefundPaymentInput["mode"],
   507→  refundMeta: NonNullable<PaymentMetadata["refund"]>,
   508→  actorUserId: string | null,
   509→  ip: string | null,
   510→  financialTransactionId: string | null
   511→): Promise<void> {
   512→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true } });
   513→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   514→
   515→  const confirmedMeta: NonNullable<PaymentMetadata["refund"]> = {
   516→    ...refundMeta,
   517→    status: "SUCCESSFUL",
   518→    confirmedAt: new Date().toISOString(),
   519→    financialTransactionId,
   520→  };
   521→
   522→  await db.$transaction(async (tx) => {
   523→    // Statut + métadonnées de remboursement de façon ATOMIQUE (même transaction)
   524→    await tx.payment.update({
   525→      where: { id: paymentId },
   526→      data: {
   527→        status: "REFUNDED",
   528→        metadata: JSON.stringify({ ...parseMetadata(payment.metadata), refund: confirmedMeta }),
   529→      },
   530→    });
   531→    // Écriture comptable REFUND (aucun doublon possible : le statut REFUNDED
   532→    // fait échouer toute nouvelle tentative de remboursement ensuite)
   533→    await tx.transaction.create({
   534→      data: {
   535→        type: "REFUND",
   536→        amount: refundMeta.amount,
   537→        reference: payment.booking.bookingReference,
   538→        description: `Remboursement billet ${payment.booking.bookingReference} (${
   539→          mode === "CASH" ? "espèces" : mode === "MOMO_REFUND" ? "MTN MoMo — remboursement" : "MTN MoMo — transfert"
   540→        }${refundMeta.msisdn ? ` vers ${refundMeta.msisdn}` : ""})`,
   541→        agencyId: payment.booking.agencyId,
   542→        paymentId: paymentId,
   543→        createdById: actorUserId,
   544→      },
   545→    });
   546→  });
   547→
   548→  await logAudit({
   549→    userId: actorUserId,
   550→    action: "REFUND_CONFIRMED",
   551→    entity: "Payment",
   552→    entityId: paymentId,
   553→    metadata: { mode, amount: refundMeta.amount, booking: payment.booking.bookingReference, financialTransactionId },
   554→    ipAddress: ip,
   555→  });
   556→}

   558→export async function initiatePaymentRefund(
   559→  paymentId: string,
   560→  input: RefundPaymentInput,
   561→  ctx: { userId: string; userName: string; ip?: string | null }
   562→): Promise<PaymentDTO> {
   563→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   564→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   565→  if (payment.status !== "SUCCESS") {
   566→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Seul un paiement réussi peut être remboursé.");
   567→  }
   568→  const meta = parseMetadata(payment.metadata);
   569→  if (meta.refund?.status === "PROCESSING" || meta.refund?.status === "SUCCESSFUL") {
   570→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un remboursement est déjà en cours ou terminé pour ce paiement.");
   571→  }
   572→  if (input.amount !== undefined && (input.amount <= 0 || input.amount > payment.amount)) {
   573→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Montant de remboursement invalide.");
   574→  }
   575→  const refundAmount = input.amount ?? payment.amount;
   576→  const bookingRef = payment.booking.bookingReference;
   577→
   578→  // ---- Espèces : finalisation immédiate ----
   579→  if (input.mode === "CASH") {
   580→    const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   581→      mode: "CASH",
   582→      status: "PROCESSING",
   583→      amount: refundAmount,
   584→      initiatedByName: ctx.userName,
   585→      initiatedAt: new Date().toISOString(),
   586→    };
   587→    await finalizeRefund(paymentId, "CASH", refundMeta, ctx.userId, ctx.ip ?? null, null);
   588→    return toPaymentDTO(await refetchPayment(paymentId));
   589→  }
   590→
   591→  // ---- Modes MoMo : disbursement obligatoire ----
   592→  if (!momoDisbursementConfigured()) {
   593→    throw new ApiError(
   594→      503,
   595→      "MOMO_NOT_CONFIGURED",
   596→      "Remboursement MTN MoMo non configuré. Renseignez les clés MOMO_DISBURSEMENT_* (voir README)."
   597→    );
   598→  }
   599→  const referenceId = randomUUID();
   600→  let msisdn: string | null = null;
   601→
   602→  if (input.mode === "MOMO_REFUND") {
   603→    if (payment.provider !== "MTN_MOMO" || !payment.providerTransactionId) {
   604→      throw new ApiError(
   605→        409,
   606→        ERROR_CODES.PAYMENT_ERROR,
   607→        "Le remboursement automatique MTN nécessite un paiement MTN MoMo (transaction d'origine requise)."
   608→      );
   609→    }
   610→    await momoRefund({
   611→      referenceId,
   612→      amount: refundAmount,
   613→      currency: momoCurrency(),
   614→      externalId: bookingRef,
   615→      referenceIdToRefund: payment.providerTransactionId,
   616→      payerMessage: `Remboursement billet ${bookingRef}`,
   617→      payeeNote: "Remboursement NZOKO TRANSPORT",
   618→    });
   619→  } else {
   620→    // MOMO_TRANSFER — envoi de fonds vers le MSISDN du passager (ou saisi)
   621→    const rawPhone: string | null = input.msisdn ?? meta.momoPhone ?? null;
   622→    if (!rawPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money du bénéficiaire requis.");
   623→    msisdn = toMomoMsisdn(rawPhone);
   624→    await momoTransfer({
   625→      referenceId,
   626→      amount: refundAmount,
   627→      currency: momoCurrency(),
   628→      externalId: bookingRef,
   629→      msisdn,
   630→      payerMessage: `Remboursement billet ${bookingRef}`,
   631→      payeeNote: "Remboursement NZOKO TRANSPORT",
   632→    });
   633→  }
   634→
   635→  const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   636→    mode: input.mode,
   637→    status: "PROCESSING",
   638→    amount: refundAmount,
   639→    referenceId,
   640→    msisdn,
   641→    initiatedByName: ctx.userName,
   642→    initiatedAt: new Date().toISOString(),
   643→  };
   644→  await writeMetadata(paymentId, (m) => ({ ...m, refund: refundMeta }));
   645→
   646→  await logAudit({
   647→    userId: ctx.userId,
   648→    action: "REFUND_INITIATED",
   649→    entity: "Payment",
   650→    entityId: paymentId,
   651→    metadata: { mode: input.mode, amount: refundAmount, booking: bookingRef, momoReferenceId: referenceId },
   652→    ipAddress: ctx.ip ?? null,
   653→  });
   654→
   655→  const fresh = await refetchPayment(paymentId);
   656→  return toPaymentDTO(fresh);
   657→}

   659→export async function pollPaymentRefundStatus(
   660→  paymentId: string,
   661→  ctx: { userId: string; ip?: string | null }
   662→): Promise<PaymentDTO> {
   663→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   664→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   665→  const meta = parseMetadata(payment.metadata);
   666→  const refund = meta.refund;
   667→  if (!refund) throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Aucun remboursement initié pour ce paiement.");
   668→  if (refund.status !== "PROCESSING" || !refund.referenceId) {
   669→    return toPaymentDTO(payment); // déjà final (SUCCESSFUL/FAILED) → idempotent
   670→  }
   671→
   672→  const status =
   673→    refund.mode === "MOMO_REFUND"
   674→      ? await momoGetRefundStatus(refund.referenceId)
   675→      : await momoGetTransferStatus(refund.referenceId);
   676→
   677→  if (status.status === "SUCCESSFUL") {
   678→    await finalizeRefund(paymentId, refund.mode, refund, ctx.userId, ctx.ip ?? null, status.financialTransactionId);
   679→  } else if (status.status === "FAILED") {
   680→    await writeMetadata(paymentId, (m) => ({
   681→      ...m,
   682→      refund: m.refund ? { ...m.refund, status: "FAILED", reason: momoFailureMessage(status), reasonCode: status.reasonCode } : m.refund,
   683→    }));
   684→    await logAudit({
   685→      userId: ctx.userId,
   686→      action: "REFUND_FAILED",
   687→      entity: "Payment",
   688→      entityId: paymentId,
   689→      metadata: { mode: refund.mode, reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   690→      ipAddress: ctx.ip ?? null,
   691→    });
   692→  }
   693→
   694→  const fresh = await refetchPayment(paymentId);
   695→  return toPaymentDTO(fresh);
   696→}

   740→export async function handleMomoWebhook(rawBody: string, ip: string | null): Promise<{ received: true; status: string }> {
   741→  const payload = safeJsonParse<{ referenceId?: string; status?: string } | null>(rawBody, null);
   742→  if (!payload?.referenceId) {
   743→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "momo callback sans referenceId" } });
   744→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload callback MoMo invalide.");
   745→  }
   746→
   747→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.referenceId } });
   748→  if (!payment) {
   749→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction MoMo inconnue.");
   750→  }
   751→
   752→  // Re-vérification systématique auprès de MTN (jamais sur parole du callback)
   753→  const dto = await pollMomoPaymentStatus(payment.id, { ip });
   754→  return { received: true, status: dto.status };
   755→}

   757→// ------------------------------------------------------------
   758→// Webhook fournisseur interne — HMAC, idempotent, jamais sur parole du client
   759→// POST /api/webhooks/payments
   760→// Signature attendue : X-Nzoko-Signature: hex HMAC-SHA256(body, WEBHOOK_SECRET)
   761→// ------------------------------------------------------------
   762→export interface WebhookPayload {
   763→  provider: string;
   764→  providerTransactionId: string;
   765→  status: "SUCCESS" | "FAILED" | "CANCELLED";
   766→  amount?: number;
   767→}
   768→
   769→export async function handlePaymentWebhook(rawBody: string, signature: string | null, ip: string | null): Promise<{ received: true; duplicate: boolean }> {
   770→  const secret = process.env.WEBHOOK_SECRET;
   771→  if (!secret) throw new ApiError(503, "WEBHOOK_NOT_CONFIGURED", "Webhook non configuré.");
   772→
   773→  if (!signature || !timingSafeEqual(signature, hmacSha256(secret, rawBody))) {
   774→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "signature invalide" } });
   775→    throw new ApiError(401, ERROR_CODES.UNAUTHORIZED, "Signature webhook invalide.");
   776→  }
   777→
   778→  const payload = safeJsonParse<WebhookPayload | null>(rawBody, null);
   779→  if (!payload?.providerTransactionId || !payload.status) {
   780→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload webhook invalide.");
   781→  }
   782→
   783→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.providerTransactionId } });
   784→  if (!payment) {
   785→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "transaction inconnue", tx: payload.providerTransactionId } });
   786→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction inconnue.");
   787→  }
   788→
   789→  if (payload.amount !== undefined && payload.amount !== payment.amount) {
   790→    await logSecurity({ event: "SUSPICIOUS", ipAddress: ip, details: { reason: "montant webhook divergent", paymentId: payment.id } });
   791→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Montant divergent — paiement bloqué.");
   792→  }
   793→
   794→  if (payload.status !== "SUCCESS") {
   795→    await db.payment.update({ where: { id: payment.id }, data: { status: payload.status } });
   796→    return { received: true, duplicate: false };
   797→  }
   798→
   799→  // Idempotence : re-confirmer un paiement déjà SUCCESS est un no-op réussi
   800→  const { alreadyConfirmed } = await confirmPaymentAndIssueTicket(payment.id, null, ip);
   801→  return { received: true, duplicate: alreadyConfirmed };
   802→}
