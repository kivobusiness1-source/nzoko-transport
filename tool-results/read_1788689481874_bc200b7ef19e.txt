    87→// --- MTN MoMo : intégration officielle (Request to Pay) ---
    88→const mtnMomoProvider: PaymentProviderInterface = {
    89→  code: "MTN_MOMO",
    90→  name: "MTN Mobile Money",
    91→  isConfigured() {
    92→    return momoCollectionConfigured();
    93→  },
    94→  async initializePayment(ctx) {
    95→    if (!this.isConfigured()) {
    96→      if (process.env.PAYMENTS_SIMULATION !== "true") {
    97→        throw new ApiError(
    98→          503,
    99→          "MOMO_NOT_CONFIGURED",
   100→          "Paiement MTN MoMo non configuré. Renseignez les clés MOMO_COLLECTION_* (voir README)."
   101→        );
   102→      }
   103→      return {
   104→        status: "PENDING",
   105→        providerTransactionId: `SIM-MTN-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`,
   106→        instructions:
   107→          "MODE DÉMONSTRATION — aucun débit réel. Utilisez le bouton « Valider (démo) » pour simuler la confirmation du fournisseur.",
   108→        isSimulation: true,
   109→      };
   110→    }
   111→    // ---- Mode PRODUCTION : Request to Pay réel ----
   112→    if (!ctx.phone) {
   113→      throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   114→    }
   115→    const msisdn = toMomoMsisdn(ctx.phone); // 400 si invalide
   116→    const referenceId = randomUUID(); // X-Reference-Id — idempotence MoMo
   117→
   118→    await momoRequestToPay({
   119→      referenceId,
   120→      amount: ctx.amount,
   121→      currency: momoCurrency(),
   122→      externalId: ctx.bookingReference ?? ctx.paymentId,
   123→      msisdn,
   124→      payerMessage: ctx.bookingReference
   125→        ? `Billet ${ctx.bookingReference} — NZOKO TRANSPORT`
   126→        : "NZOKO TRANSPORT",
   127→      payeeNote: `Paiement billet ${ctx.bookingReference ?? ""}`.trim(),
   128→    });
   129→
   130→    return {
   131→      status: "PROCESSING",
   132→      providerTransactionId: referenceId,
   133→      instructions:
   134→        "Demande de paiement envoyée sur votre téléphone. Validez-la depuis votre application MTN MoMo ou le menu MoMo de votre SIM, puis patientez quelques secondes.",
   135→      isSimulation: false,
   136→    };
   137→  },
   138→  async verifyPayment(providerTransactionId) {
   139→    if (providerTransactionId.startsWith("SIM-")) return "SUCCESS"; // démo uniquement
   140→    if (!this.isConfigured()) return "PROCESSING";
   141→    const st = await momoGetRequestToPayStatus(providerTransactionId);
   142→    return st.status === "SUCCESSFUL" ? "SUCCESS" : st.status === "FAILED" ? "FAILED" : "PROCESSING";
   143→  },
   144→  async refundPayment(providerTransactionId) {
   145→    if (!momoDisbursementConfigured()) return false;
   146→    await momoRefund({
   147→      referenceId: randomUUID(),
   148→      amount: 0, // remplacé par le contexte réel dans initiatePaymentRefund
   149→      currency: momoCurrency(),
   150→      externalId: providerTransactionId,
   151→      referenceIdToRefund: providerTransactionId,
   152→    });
   153→    return true; // initiation acceptée — suivi par pollPaymentRefundStatus
   154→  },

   280→async function writeMetadata(paymentId: string, mutate: (meta: PaymentMetadata) => PaymentMetadata): Promise<void> {
   281→  const payment = await db.payment.findUnique({ where: { id: paymentId }, select: { metadata: true } });
   282→  if (!payment) return;
   283→  const next = mutate(parseMetadata(payment.metadata));
   284→  await db.payment.update({ where: { id: paymentId }, data: { metadata: JSON.stringify(next) } });
   285→}

   291→export async function createPayment(
   292→  input: PayBookingInput,
   293→  ctx: { actorUserId?: string | null; ip?: string | null }
   294→): Promise<PaymentDTO> {
   295→  const booking = await db.booking.findUnique({
   296→    where: { id: input.bookingId },
   297→    include: { trip: true, payment: true },
   298→  });
   299→  if (!booking) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Réservation introuvable.");
   300→  if (booking.status !== "PENDING") {
   301→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette réservation n'est plus en attente de paiement.");
   302→  }
   303→  if (booking.expiresAt && booking.expiresAt < new Date()) {
   304→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le délai de réservation a expiré. Veuillez réserver à nouveau.");
   305→  }
   306→  if (booking.payment.some((p) => ["PENDING", "PROCESSING", "SUCCESS"].includes(p.status))) {
   307→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un paiement est déjà en cours pour cette réservation.");
   308→  }
   309→
   310→  const provider = getProvider(input.provider);
   311→
   312→  // MTN MoMo réel : validation E.164 AVANT toute écriture (400 propre, aucune ligne créée)
   313→  let momoPhone = input.momoPhone;
   314→  if (input.provider === "MTN_MOMO" && provider.isConfigured()) {
   315→    if (!momoPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   316→    momoPhone = toMomoMsisdn(momoPhone);
   317→  }
   318→
   319→  const payment = await db.payment.create({
   320→    data: {
   321→      bookingId: booking.id,
   322→      provider: input.provider,
   323→      amount: booking.amount,
   324→      status: "PENDING",
   325→      createdById: ctx.actorUserId ?? null,
   326→    },
   327→  });
   328→
   329→  let result: ProviderResult;
   330→  try {
   331→    result = await provider.initializePayment({
   332→      paymentId: payment.id,
   333→      amount: booking.amount,
   334→      phone: momoPhone,
   335→      senderName: input.senderName,
   336→      bookingReference: booking.bookingReference,
   337→    });
   338→  } catch (err) {
   339→    // Échec d'initialisation fournisseur (ex : numéro MoMo introuvable) →
   340→    // le paiement est marqué FAILED (réessayable) et l'erreur remonte au client.
   341→    const reason = err instanceof ApiError ? err.message : "Initialisation du paiement impossible.";
   342→    await db.payment.update({
   343→      where: { id: payment.id },
   344→      data: { status: "FAILED", metadata: JSON.stringify({ isSimulation: false, reason }) },
   345→    });
   346→    throw err;
   347→  }
   348→
   349→  const momoMeta: PaymentMetadata["momo"] | undefined =
   350→    input.provider === "MTN_MOMO" && !result.isSimulation && result.providerTransactionId
   351→      ? {
   352→          product: "collection",
   353→          referenceId: result.providerTransactionId,
   354→          msisdn: momoPhone ?? "",
   355→          currency: momoCurrency(),
   356→          environment: momoConfigStatus().environment,
   357→        }
   358→      : undefined;
   359→
   360→  const updated = await db.payment.update({
   361→    where: { id: payment.id },
   362→    data: {
   363→      status: result.status,
   364→      providerTransactionId: result.providerTransactionId,
   365→      metadata: JSON.stringify({
   366→        isSimulation: result.isSimulation,
   367→        instructions: result.instructions,
   368→        momoPhone: momoPhone ?? null,
   369→        senderName: input.senderName ?? null,
   370→        ip: ctx.ip ?? null,
   371→        ...(momoMeta ? { momo: momoMeta } : {}),
   372→      }),
   373→    },
   374→    include: { createdBy: true, booking: true },
   375→  });
   376→
   377→  await logAudit({
   378→    userId: ctx.actorUserId ?? null,
   379→    action: "PAYMENT_INITIALIZED",
   380→    entity: "Payment",
   381→    entityId: payment.id,
   382→    metadata: {
   383→      provider: input.provider,
   384→      amount: booking.amount,
   385→      simulation: result.isSimulation,
   386→      ...(result.isSimulation ? {} : { momoReferenceId: result.providerTransactionId }),
   387→    },
   388→    ipAddress: ctx.ip ?? null,
   389→  });
   390→
   391→  return toPaymentDTO(updated);
   392→}

   394→// ------------------------------------------------------------
   395→// Confirmation idempotente + émission billet
   396→// Cœur du système : appelée par cash, simulation, webhook, confirmation manuelle.

   398→
   399→export async function confirmPaymentAndIssueTicket(
   400→  paymentId: string,
   401→  actorUserId: string | null,
   402→  ip: string | null
   403→): Promise<{ payment: PaymentDTO; alreadyConfirmed: boolean }> {
   404→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   405→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   406→
   407→  if (payment.status === "SUCCESS") {
   408→    return { payment: toPaymentDTO(payment), alreadyConfirmed: true }; // idempotence stricte
   409→  }
   410→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   411→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Ce paiement ne peut plus être confirmé.");
   412→  }
   413→
   414→  await db.$transaction(async (tx) => {
   415→    // Re-vérification atomique du statut dans la transaction
   416→    const fresh = await tx.payment.findUnique({ where: { id: payment.id }, select: { status: true } });
   417→    if (!fresh || fresh.status !== payment.status) {
   418→      throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le paiement a changé d'état concurrentiellement.");
   419→    }
   420→
   421→    await tx.payment.update({ where: { id: payment.id }, data: { status: "SUCCESS" } });
   422→    await tx.booking.update({ where: { id: payment.bookingId }, data: { status: "CONFIRMED" } });
   423→    await tx.seatOccupancy.updateMany({
   424→      where: { bookingId: payment.bookingId },
   425→      data: { status: "BOOKED", expiresAt: null },
   426→    });
   427→
   428→    // Billet émis exactement une fois (contrainte unique bookingId)
   429→    await issueTicketForBooking(tx, payment.bookingId, payment.booking.bookingReference);
   430→
   431→    // Écriture comptable idempotente (référence unique)
   432→    await tx.transaction.create({
   433→      data: {
   434→        type: "INCOME",
   435→        amount: payment.amount,
   436→        reference: payment.booking.bookingReference,
   437→        description: `Vente billet ${payment.booking.bookingReference} (${payment.provider})`,
   438→        agencyId: payment.booking.agencyId,
   439→        paymentId: payment.id,
   440→        createdById: actorUserId,
   441→      },
   442→    });
   443→  });
   444→
   445→  await logAudit({
   446→    userId: actorUserId,
   447→    action: "PAYMENT_CONFIRMED",
   448→    entity: "Payment",
   449→    entityId: payment.id,
   450→    metadata: { provider: payment.provider, amount: payment.amount, booking: payment.booking.bookingReference },
   451→    ipAddress: ip,
   452→  });
   453→
   454→  const final = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   455→  return { payment: toPaymentDTO(final!), alreadyConfirmed: false };
   456→}

   459→export async function confirmCashPayment(paymentId: string, actorUserId: string, ip: string | null): Promise<PaymentDTO> {
   460→  const payment = await db.payment.findUnique({ where: { id: paymentId } });
   461→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");

   465→  const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, actorUserId, ip);
   466→  return dto;

   480→// ------------------------------------------------------------
   481→// MTN MoMo — suivi Request to Pay (polling public, idempotent)
   482→// Référence : GET /collection/v1_0/requesttopay/{referenceId}
   483→// ------------------------------------------------------------
   484→
   485→export async function pollMomoPaymentStatus(
   486→  paymentId: string,
   487→  ctx: { actorUserId?: string | null; ip?: string | null }
   488→): Promise<PaymentDTO> {
   489→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   490→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   491→  if (payment.provider !== "MTN_MOMO") {
   492→    throw new ApiError(400, ERROR_CODES.BAD_REQUEST, "Ce paiement n'est pas un paiement MTN Mobile Money.");
   493→  }
   494→  // Statut déjà final → réponse idempotente (pas d'appel MTN inutile)
   495→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   496→    return toPaymentDTO(payment);
   497→  }
   498→  if (!payment.providerTransactionId || payment.providerTransactionId.startsWith("SIM-")) {
   499→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Paiement de démonstration — utilisez la validation démo.");
   500→  }
   501→
   502→  const status = await momoGetRequestToPayStatus(payment.providerTransactionId);
   503→
   504→  if (status.status === "SUCCESSFUL") {
   505→    // Confirmation idempotente (billet + écriture comptable) — le cœur existant
   506→    const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, ctx.actorUserId ?? null, ctx.ip ?? null);
   507→    await writeMetadata(paymentId, (m) => ({
   508→      ...m,
   509→      momo: m.momo
   510→        ? { ...m.momo, financialTransactionId: status.financialTransactionId }
   511→        : m.momo,
   512→    }));
   513→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   514→    return toPaymentDTO(fresh!);
   515→  }
   516→
   517→  if (status.status === "FAILED") {
   518→    await db.payment.update({ where: { id: paymentId }, data: { status: "FAILED" } });
   519→    await writeMetadata(paymentId, (m) => ({
   520→      ...m,
   521→      reason: momoFailureMessage(status),
   522→      momo: m.momo ? { ...m.momo, reasonCode: status.reasonCode, reason: status.reasonMessage } : m.momo,
   523→    }));
   524→    await logAudit({
   525→      userId: ctx.actorUserId ?? null,
   526→      action: "MOMO_PAYMENT_FAILED",
   527→      entity: "Payment",
   528→      entityId: paymentId,
   529→      metadata: { reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   530→      ipAddress: ctx.ip ?? null,
   531→    });
   532→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   533→    return toPaymentDTO(fresh!);
   534→  }
   535→
   536→  // Toujours PENDING côté MTN
   537→  return toPaymentDTO(payment);
   538→}

   540→// ------------------------------------------------------------
   541→// MTN MoMo — REMBOURSEMENTS (envoi de fonds aux clients)
   542→// Modes : MOMO_REFUND (Refund API, référence R2P d'origine),
   543→//         MOMO_TRANSFER (Transfer vers un MSISDN),
   544→//         CASH (espèces rendues au guichet — finalisation immédiate).
   545→// ------------------------------------------------------------
   546→
   547→async function finalizeRefund(
   548→  paymentId: string,
   549→  mode: RefundPaymentInput["mode"],
   550→  refundMeta: NonNullable<PaymentMetadata["refund"]>,
   551→  actorUserId: string | null,
   552→  ip: string | null,
   553→  financialTransactionId: string | null
   554→): Promise<void> {
   555→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true } });
   556→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   557→
   558→  const confirmedMeta: NonNullable<PaymentMetadata["refund"]> = {
   559→    ...refundMeta,
   560→    status: "SUCCESSFUL",
   561→    confirmedAt: new Date().toISOString(),
   562→    financialTransactionId,
   563→  };
   564→
   565→  await db.$transaction(async (tx) => {
   566→    // Statut + métadonnées de remboursement de façon ATOMIQUE (même transaction)
   567→    await tx.payment.update({
   568→      where: { id: paymentId },
   569→      data: {
   570→        status: "REFUNDED",
   571→        metadata: JSON.stringify({ ...parseMetadata(payment.metadata), refund: confirmedMeta }),
   572→      },
   573→    });
   574→    // Écriture comptable REFUND (aucun doublon possible : le statut REFUNDED
   575→    // fait échouer toute nouvelle tentative de remboursement ensuite)
   576→    await tx.transaction.create({
   577→      data: {
   578→        type: "REFUND",
   579→        amount: refundMeta.amount,
   580→        reference: payment.booking.bookingReference,
   581→        description: `Remboursement billet ${payment.booking.bookingReference} (${
   582→          mode === "CASH" ? "espèces" : mode === "MOMO_REFUND" ? "MTN MoMo — remboursement" : "MTN MoMo — transfert"
   583→        }${refundMeta.msisdn ? ` vers ${refundMeta.msisdn}` : ""})`,
   584→        agencyId: payment.booking.agencyId,
   585→        paymentId: paymentId,
   586→        createdById: actorUserId,
   587→      },
   588→    });
   589→  });
   590→
   591→  await logAudit({
   592→    userId: actorUserId,
   593→    action: "REFUND_CONFIRMED",
   594→    entity: "Payment",
   595→    entityId: paymentId,
   596→    metadata: { mode, amount: refundMeta.amount, booking: payment.booking.bookingReference, financialTransactionId },
   597→    ipAddress: ip,
   598→  });
   599→}

   601→export async function initiatePaymentRefund(
   602→  paymentId: string,
   603→  input: RefundPaymentInput,
   604→  ctx: { userId: string; userName: string; ip?: string | null }
   605→): Promise<PaymentDTO> {
   606→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   607→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   608→  if (payment.status !== "SUCCESS") {
   609→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Seul un paiement réussi peut être remboursé.");
   610→  }
   611→  const meta = parseMetadata(payment.metadata);
   612→  if (meta.refund?.status === "PROCESSING" || meta.refund?.status === "SUCCESSFUL") {
   613→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un remboursement est déjà en cours ou terminé pour ce paiement.");
   614→  }
   615→  if (input.amount !== undefined && (input.amount <= 0 || input.amount > payment.amount)) {
   616→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Montant de remboursement invalide.");
   617→  }
   618→  const refundAmount = input.amount ?? payment.amount;
   619→  const bookingRef = payment.booking.bookingReference;
   620→
   621→  // ---- Espèces : finalisation immédiate ----
   622→  if (input.mode === "CASH") {
   623→    const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   624→      mode: "CASH",
   625→      status: "PROCESSING",
   626→      amount: refundAmount,
   627→      initiatedByName: ctx.userName,
   628→      initiatedAt: new Date().toISOString(),
   629→    };
   630→    await finalizeRefund(paymentId, "CASH", refundMeta, ctx.userId, ctx.ip ?? null, null);
   631→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   632→    return toPaymentDTO(fresh!);
   633→  }
   634→
   635→  // ---- Modes MoMo : disbursement obligatoire ----
   636→  if (!momoDisbursementConfigured()) {
   637→    throw new ApiError(
   638→      503,
   639→      "MOMO_NOT_CONFIGURED",
   640→      "Remboursement MTN MoMo non configuré. Renseignez les clés MOMO_DISBURSEMENT_* (voir README)."
   641→    );
   642→  }
   643→  const referenceId = randomUUID();
   644→  let msisdn: string | null = null;
   645→
   646→  if (input.mode === "MOMO_REFUND") {
   647→    if (payment.provider !== "MTN_MOMO" || !payment.providerTransactionId || payment.providerTransactionId.startsWith("SIM-")) {
   648→      throw new ApiError(
   649→        409,
   650→        ERROR_CODES.PAYMENT_ERROR,
   651→        "Le remboursement automatique MTN nécessite un paiement MTN MoMo réel (pas une simulation)."
   652→      );
   653→    }
   654→    await momoRefund({
   655→      referenceId,
   656→      amount: refundAmount,
   657→      currency: momoCurrency(),
   658→      externalId: bookingRef,
   659→      referenceIdToRefund: payment.providerTransactionId,
   660→      payerMessage: `Remboursement billet ${bookingRef}`,
   661→      payeeNote: "Remboursement NZOKO TRANSPORT",
   662→    });
   663→  } else {
   664→    // MOMO_TRANSFER — envoi de fonds vers le MSISDN du passager (ou saisi)
   665→    const rawPhone: string | null = input.msisdn ?? meta.momoPhone ?? null;
   666→    if (!rawPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money du bénéficiaire requis.");
   667→    msisdn = toMomoMsisdn(rawPhone);
   668→    await momoTransfer({
   669→      referenceId,
   670→      amount: refundAmount,
   671→      currency: momoCurrency(),
   672→      externalId: bookingRef,
   673→      msisdn,
   674→      payerMessage: `Remboursement billet ${bookingRef}`,
   675→      payeeNote: "Remboursement NZOKO TRANSPORT",
   676→    });
   677→  }
   678→
   679→  const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   680→    mode: input.mode,
   681→    status: "PROCESSING",
   682→    amount: refundAmount,
   683→    referenceId,
   684→    msisdn,
   685→    initiatedByName: ctx.userName,
   686→    initiatedAt: new Date().toISOString(),
   687→  };
   688→  await writeMetadata(paymentId, (m) => ({ ...m, refund: refundMeta }));
   689→
   690→  await logAudit({
   691→    userId: ctx.userId,
   692→    action: "REFUND_INITIATED",
   693→    entity: "Payment",
   694→    entityId: paymentId,
   695→    metadata: { mode: input.mode, amount: refundAmount, booking: bookingRef, momoReferenceId: referenceId },
   696→    ipAddress: ctx.ip ?? null,
   697→  });
   698→
   699→  const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   700→  return toPaymentDTO(fresh!);
   701→}

   703→export async function pollPaymentRefundStatus(
   704→  paymentId: string,
   705→  ctx: { userId: string; ip?: string | null }
   706→): Promise<PaymentDTO> {
   707→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   708→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   709→  const meta = parseMetadata(payment.metadata);
   710→  const refund = meta.refund;
   711→  if (!refund) throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Aucun remboursement initié pour ce paiement.");
   712→  if (refund.status !== "PROCESSING" || !refund.referenceId) {
   713→    return toPaymentDTO(payment); // déjà final (SUCCESSFUL/FAILED) → idempotent
   714→  }
   715→
   716→  const status =
   717→    refund.mode === "MOMO_REFUND"
   718→      ? await momoGetRefundStatus(refund.referenceId)
   719→      : await momoGetTransferStatus(refund.referenceId);
   720→
   721→  if (status.status === "SUCCESSFUL") {
   722→    await finalizeRefund(paymentId, refund.mode, refund, ctx.userId, ctx.ip ?? null, status.financialTransactionId);
   723→  } else if (status.status === "FAILED") {
   724→    await writeMetadata(paymentId, (m) => ({
   725→      ...m,
   726→      refund: m.refund ? { ...m.refund, status: "FAILED", reason: momoFailureMessage(status), reasonCode: status.reasonCode } : m.refund,
   727→    }));
   728→    await logAudit({
   729→      userId: ctx.userId,
   730→      action: "REFUND_FAILED",
   731→      entity: "Payment",
   732→      entityId: paymentId,
   733→      metadata: { mode: refund.mode, reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   734→      ipAddress: ctx.ip ?? null,
   735→    });
   736→  }
   737→
   738→  const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   739→  return toPaymentDTO(fresh!);
   740→}

   784→export async function handleMomoWebhook(rawBody: string, ip: string | null): Promise<{ received: true; status: string }> {
   785→  const payload = safeJsonParse<{ referenceId?: string; status?: string } | null>(rawBody, null);
   786→  if (!payload?.referenceId) {
   787→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "momo callback sans referenceId" } });
   788→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload callback MoMo invalide.");
   789→  }
   790→
   791→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.referenceId } });
   792→  if (!payment) {
   793→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction MoMo inconnue.");
   794→  }
   795→
   796→  // Re-vérification systématique auprès de MTN (jamais sur parole du callback)
   797→  const dto = await pollMomoPaymentStatus(payment.id, { ip });
   798→  return { received: true, status: dto.status };
   799→}

   801→// ------------------------------------------------------------
   802→// Webhook fournisseur interne — HMAC, idempotent, jamais sur parole du client
   803→// POST /api/webhooks/payments
   804→// Signature attendue : X-Nzoko-Signature: hex HMAC-SHA256(body, WEBHOOK_SECRET)
   805→// ------------------------------------------------------------
   806→export interface WebhookPayload {
   807→  provider: string;
   808→  providerTransactionId: string;
   809→  status: "SUCCESS" | "FAILED" | "CANCELLED";
   810→  amount?: number;
   811→}
   812→
   813→export async function handlePaymentWebhook(rawBody: string, signature: string | null, ip: string | null): Promise<{ received: true; duplicate: boolean }> {
   814→  const secret = process.env.WEBHOOK_SECRET;
   815→  if (!secret) throw new ApiError(503, "WEBHOOK_NOT_CONFIGURED", "Webhook non configuré.");
   816→
   817→  if (!signature || !timingSafeEqual(signature, hmacSha256(secret, rawBody))) {
   818→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "signature invalide" } });
   819→    throw new ApiError(401, ERROR_CODES.UNAUTHORIZED, "Signature webhook invalide.");
   820→  }
   821→
   822→  const payload = safeJsonParse<WebhookPayload | null>(rawBody, null);
   823→  if (!payload?.providerTransactionId || !payload.status) {
   824→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload webhook invalide.");
   825→  }
   826→
   827→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.providerTransactionId } });
   828→  if (!payment) {
   829→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "transaction inconnue", tx: payload.providerTransactionId } });
   830→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction inconnue.");
   831→  }
   832→
   833→  if (payload.amount !== undefined && payload.amount !== payment.amount) {
   834→    await logSecurity({ event: "SUSPICIOUS", ipAddress: ip, details: { reason: "montant webhook divergent", paymentId: payment.id } });
   835→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Montant divergent — paiement bloqué.");
   836→  }
   837→
   838→  if (payload.status !== "SUCCESS") {
   839→    await db.payment.update({ where: { id: payment.id }, data: { status: payload.status } });
   840→    return { received: true, duplicate: false };
   841→  }
   842→
   843→  // Idempotence : re-confirmer un paiement déjà SUCCESS est un no-op réussi
   844→  const { alreadyConfirmed } = await confirmPaymentAndIssueTicket(payment.id, null, ip);
   845→  return { received: true, duplicate: alreadyConfirmed };
   846→}
