    91→function ref(len: number): string {
    92→  const alphabet = "ABCDEFGHJKMNPQRSTUVWXYZ23456789";
    93→  const bytes = crypto.randomBytes(len);
    94→  let out = "";
    95→  for (let i = 0; i < len; i++) out += alphabet[bytes[i] % alphabet.length];
    96→  return out;
    97→}
