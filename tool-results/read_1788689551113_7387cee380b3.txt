   291→   291→export async function createPayment(
   292→   292→  input: PayBookingInput,
   293→   293→  ctx: { actorUserId?: string | null; ip?: string | null }
   294→   294→): Promise<PaymentDTO> {
   295→   295→  const booking = await db.booking.findUnique({
   296→   296→    where: { id: input.bookingId },
   297→   297→    include: { trip: true, payment: true },
   298→   298→  });
   299→   299→  if (!booking) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Réservation introuvable.");
   300→   300→  if (booking.status !== "PENDING") {
   301→   301→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Cette réservation n'est plus en attente de paiement.");
   302→   302→  }
   303→   303→  if (booking.expiresAt && booking.expiresAt < new Date()) {
   304→   304→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le délai de réservation a expiré. Veuillez réserver à nouveau.");
   305→   305→  }
   306→   306→  if (booking.payment.some((p) => ["PENDING", "PROCESSING", "SUCCESS"].includes(p.status))) {
   307→   307→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un paiement est déjà en cours pour cette réservation.");
   308→   308→  }
   309→   309→
   310→   310→  const provider = getProvider(input.provider);
   311→   311→
   312→   312→  // MTN MoMo réel : validation E.164 AVANT toute écriture (400 propre, aucune ligne créée)
   313→   313→  let momoPhone = input.momoPhone;
   314→   314→  if (input.provider === "MTN_MOMO" && provider.isConfigured()) {
   315→   315→    if (!momoPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money requis.");
   316→   316→    momoPhone = toMomoMsisdn(momoPhone);
   317→   317→  }
   318→   318→
   319→   319→  const payment = await db.payment.create({
   320→   320→    data: {
   321→   321→      bookingId: booking.id,
   322→   322→      provider: input.provider,
   323→   323→      amount: booking.amount,
   324→   324→      status: "PENDING",
   325→   325→      createdById: ctx.actorUserId ?? null,
   326→   326→    },
   327→   327→  });
   328→   328→
   329→   329→  let result: ProviderResult;
   330→   330→  try {
   331→   331→    result = await provider.initializePayment({
   332→   332→      paymentId: payment.id,
   333→   333→      amount: booking.amount,
   334→   334→      phone: momoPhone,
   335→   335→      senderName: input.senderName,
   336→   336→      bookingReference: booking.bookingReference,
   337→   337→    });
   338→   338→  } catch (err) {
   339→   339→    // Échec d'initialisation fournisseur (ex : numéro MoMo introuvable) →
   340→   340→    // le paiement est marqué FAILED (réessayable) et l'erreur remonte au client.
   341→   341→    const reason = err instanceof ApiError ? err.message : "Initialisation du paiement impossible.";
   342→   342→    await db.payment.update({
   343→   343→      where: { id: payment.id },
   344→   344→      data: { status: "FAILED", metadata: JSON.stringify({ isSimulation: false, reason }) },
   345→   345→    });
   346→   346→    throw err;
   347→   347→  }
   348→   348→
   349→   349→  const momoMeta: PaymentMetadata["momo"] | undefined =
   350→   350→    input.provider === "MTN_MOMO" && !result.isSimulation && result.providerTransactionId
   351→   351→      ? {
   352→   352→          product: "collection",
   353→   353→          referenceId: result.providerTransactionId,
   354→   354→          msisdn: momoPhone ?? "",
   355→   355→          currency: momoCurrency(),
   356→   356→          environment: momoConfigStatus().environment,
   357→   357→        }
   358→   358→      : undefined;
   359→   359→
   360→   360→  const updated = await db.payment.update({
   361→   361→    where: { id: payment.id },
   362→   362→    data: {
   363→   363→      status: result.status,
   364→   364→      providerTransactionId: result.providerTransactionId,
   365→   365→      metadata: JSON.stringify({
   366→   366→        isSimulation: result.isSimulation,
   367→   367→        instructions: result.instructions,
   368→   368→        momoPhone: momoPhone ?? null,
   369→   369→        senderName: input.senderName ?? null,
   370→   370→        ip: ctx.ip ?? null,
   371→   371→        ...(momoMeta ? { momo: momoMeta } : {}),
   372→   372→      }),
   373→   373→    },
   374→   374→    include: { createdBy: true, booking: true },
   375→   375→  });
   376→   376→
   377→   377→  await logAudit({
   378→   378→    userId: ctx.actorUserId ?? null,
   379→   379→    action: "PAYMENT_INITIALIZED",
   380→   380→    entity: "Payment",
   381→   381→    entityId: payment.id,
   382→   382→    metadata: {
   383→   383→      provider: input.provider,
   384→   384→      amount: booking.amount,
   385→   385→      simulation: result.isSimulation,
   386→   386→      ...(result.isSimulation ? {} : { momoReferenceId: result.providerTransactionId }),
   387→   387→    },
   388→   388→    ipAddress: ctx.ip ?? null,
   389→   389→  });
   390→   390→
   391→   391→  return toPaymentDTO(updated);
   392→   392→}

   394→   394→// ------------------------------------------------------------
   395→   395→// Confirmation idempotente + émission billet
   396→   396→// Cœur du système : appelée par cash, simulation, webhook, confirmation manuelle.
   397→   397→// ------------------------------------------------------------
   398→   398→
   399→   399→export async function confirmPaymentAndIssueTicket(
   400→   400→  paymentId: string,
   401→   401→  actorUserId: string | null,
   402→   402→  ip: string | null
   403→   403→): Promise<{ payment: PaymentDTO; alreadyConfirmed: boolean }> {
   404→   404→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   405→   405→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   406→   406→
   407→   407→  if (payment.status === "SUCCESS") {
   408→   408→    return { payment: toPaymentDTO(payment), alreadyConfirmed: true }; // idempotence stricte
   409→   409→  }
   410→   410→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   411→   411→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Ce paiement ne peut plus être confirmé.");
   412→   412→  }
   413→   413→
   414→   414→  await db.$transaction(async (tx) => {
   415→   415→    // Re-vérification atomique du statut dans la transaction
   416→   416→    const fresh = await tx.payment.findUnique({ where: { id: payment.id }, select: { status: true } });
   417→   417→    if (!fresh || fresh.status !== payment.status) {
   418→   418→      throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Le paiement a changé d'état concurrentiellement.");
   419→   419→    }
   420→   420→
   421→   421→    await tx.payment.update({ where: { id: payment.id }, data: { status: "SUCCESS" } });
   422→   422→    await tx.booking.update({ where: { id: payment.bookingId }, data: { status: "CONFIRMED" } });
   423→   423→    await tx.seatOccupancy.updateMany({
   424→   424→      where: { bookingId: payment.bookingId },
   425→   425→      data: { status: "BOOKED", expiresAt: null },
   426→   426→    });
   427→   427→
   428→   428→    // Billet émis exactement une fois (contrainte unique bookingId)
   429→   429→    await issueTicketForBooking(tx, payment.bookingId, payment.booking.bookingReference);
   430→   430→
   431→   431→    // Écriture comptable idempotente (référence unique)
   432→   432→    await tx.transaction.create({
   433→   433→      data: {
   434→   434→        type: "INCOME",
   435→   435→        amount: payment.amount,
   436→   436→        reference: payment.booking.bookingReference,
   437→   437→        description: `Vente billet ${payment.booking.bookingReference} (${payment.provider})`,
   438→   438→        agencyId: payment.booking.agencyId,
   439→   439→        paymentId: payment.id,
   440→   440→        createdById: actorUserId,
   441→   441→      },
   442→   442→    });
   443→   443→  });
   444→   444→
   445→   445→  await logAudit({
   446→   446→    userId: actorUserId,
   447→   447→    action: "PAYMENT_CONFIRMED",
   448→   448→    entity: "Payment",
   449→   449→    entityId: payment.id,
   450→   450→    metadata: { provider: payment.provider, amount: payment.amount, booking: payment.booking.bookingReference },
   451→   451→    ipAddress: ip,

   485→   485→export async function pollMomoPaymentStatus(
   486→   486→  paymentId: string,
   487→   487→  ctx: { actorUserId?: string | null; ip?: string | null }
   488→   488→): Promise<PaymentDTO> {
   489→   489→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   490→   490→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   491→   491→  if (payment.provider !== "MTN_MOMO") {
   492→   492→    throw new ApiError(400, ERROR_CODES.BAD_REQUEST, "Ce paiement n'est pas un paiement MTN Mobile Money.");
   493→   493→  }
   494→   494→  // Statut déjà final → réponse idempotente (pas d'appel MTN inutile)
   495→   495→  if (payment.status !== "PENDING" && payment.status !== "PROCESSING") {
   496→   496→    return toPaymentDTO(payment);
   497→   497→  }
   498→   498→  if (!payment.providerTransactionId || payment.providerTransactionId.startsWith("SIM-")) {
   499→   499→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Paiement de démonstration — utilisez la validation démo.");
   500→   500→  }
   501→   501→
   502→   502→  const status = await momoGetRequestToPayStatus(payment.providerTransactionId);
   503→   503→
   504→   504→  if (status.status === "SUCCESSFUL") {
   505→   505→    // Confirmation idempotente (billet + écriture comptable) — le cœur existant
   506→   506→    const { payment: dto } = await confirmPaymentAndIssueTicket(paymentId, ctx.actorUserId ?? null, ctx.ip ?? null);
   507→   507→    await writeMetadata(paymentId, (m) => ({
   508→   508→      ...m,
   509→   509→      momo: m.momo
   510→   510→        ? { ...m.momo, financialTransactionId: status.financialTransactionId }
   511→   511→        : m.momo,
   512→   512→    }));
   513→   513→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   514→   514→    return toPaymentDTO(fresh!);
   515→   515→  }
   516→   516→
   517→   517→  if (status.status === "FAILED") {
   518→   518→    await db.payment.update({ where: { id: paymentId }, data: { status: "FAILED" } });
   519→   519→    await writeMetadata(paymentId, (m) => ({
   520→   520→      ...m,
   521→   521→      reason: momoFailureMessage(status),
   522→   522→      momo: m.momo ? { ...m.momo, reasonCode: status.reasonCode, reason: status.reasonMessage } : m.momo,
   523→   523→    }));
   524→   524→    await logAudit({
   525→   525→      userId: ctx.actorUserId ?? null,
   526→   526→      action: "MOMO_PAYMENT_FAILED",
   527→   527→      entity: "Payment",
   528→   528→      entityId: paymentId,
   529→   529→      metadata: { reasonCode: status.reasonCode, booking: payment.booking.bookingReference },
   530→   530→      ipAddress: ctx.ip ?? null,
   531→   531→    });
   532→   532→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   533→   533→    return toPaymentDTO(fresh!);
   534→   534→  }
   535→   535→
   536→   536→  // Toujours PENDING côté MTN
   537→   537→  return toPaymentDTO(payment);
   538→   538→}

   543→   543→//         MOMO_TRANSFER (Transfer vers un MSISDN),
   544→   544→//         CASH (espèces rendues au guichet — finalisation immédiate).

   546→   546→
   547→   547→async function finalizeRefund(
   548→   548→  paymentId: string,
   549→   549→  mode: RefundPaymentInput["mode"],
   550→   550→  refundMeta: NonNullable<PaymentMetadata["refund"]>,
   551→   551→  actorUserId: string | null,
   552→   552→  ip: string | null,
   553→   553→  financialTransactionId: string | null
   554→   554→): Promise<void> {
   555→   555→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true } });
   556→   556→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   557→   557→
   558→   558→  const confirmedMeta: NonNullable<PaymentMetadata["refund"]> = {
   559→   559→    ...refundMeta,
   560→   560→    status: "SUCCESSFUL",
   561→   561→    confirmedAt: new Date().toISOString(),
   562→   562→    financialTransactionId,
   563→   563→  };
   564→   564→
   565→   565→  await db.$transaction(async (tx) => {
   566→   566→    // Statut + métadonnées de remboursement de façon ATOMIQUE (même transaction)
   567→   567→    await tx.payment.update({
   568→   568→      where: { id: paymentId },
   569→   569→      data: {
   570→   570→        status: "REFUNDED",
   571→   571→        metadata: JSON.stringify({ ...parseMetadata(payment.metadata), refund: confirmedMeta }),
   572→   572→      },
   573→   573→    });
   574→   574→    // Écriture comptable REFUND (aucun doublon possible : le statut REFUNDED
   575→   575→    // fait échouer toute nouvelle tentative de remboursement ensuite)
   576→   576→    await tx.transaction.create({
   577→   577→      data: {
   578→   578→        type: "REFUND",
   579→   579→        amount: refundMeta.amount,
   580→   580→        reference: payment.booking.bookingReference,
   581→   581→        description: `Remboursement billet ${payment.booking.bookingReference} (${
   582→   582→          mode === "CASH" ? "espèces" : mode === "MOMO_REFUND" ? "MTN MoMo — remboursement" : "MTN MoMo — transfert"
   583→   583→        }${refundMeta.msisdn ? ` vers ${refundMeta.msisdn}` : ""})`,
   584→   584→        agencyId: payment.booking.agencyId,
   585→   585→        paymentId: paymentId,
   586→   586→        createdById: actorUserId,
   587→   587→      },
   588→   588→    });
   589→   589→  });
   590→   590→
   591→   591→  await logAudit({
   592→   592→    userId: actorUserId,
   593→   593→    action: "REFUND_CONFIRMED",
   594→   594→    entity: "Payment",
   595→   595→    entityId: paymentId,
   596→   596→    metadata: { mode, amount: refundMeta.amount, booking: payment.booking.bookingReference, financialTransactionId },
   597→   597→    ipAddress: ip,
   598→   598→  });
   599→   599→}

   601→   601→export async function initiatePaymentRefund(
   602→   602→  paymentId: string,
   603→   603→  input: RefundPaymentInput,
   604→   604→  ctx: { userId: string; userName: string; ip?: string | null }
   605→   605→): Promise<PaymentDTO> {
   606→   606→  const payment = await db.payment.findUnique({ where: { id: paymentId }, include: { booking: true, createdBy: true } });
   607→   607→  if (!payment) throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Paiement introuvable.");
   608→   608→  if (payment.status !== "SUCCESS") {
   609→   609→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Seul un paiement réussi peut être remboursé.");
   610→   610→  }
   611→   611→  const meta = parseMetadata(payment.metadata);
   612→   612→  if (meta.refund?.status === "PROCESSING" || meta.refund?.status === "SUCCESSFUL") {
   613→   613→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Un remboursement est déjà en cours ou terminé pour ce paiement.");
   614→   614→  }
   615→   615→  if (input.amount !== undefined && (input.amount <= 0 || input.amount > payment.amount)) {
   616→   616→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Montant de remboursement invalide.");
   617→   617→  }
   618→   618→  const refundAmount = input.amount ?? payment.amount;
   619→   619→  const bookingRef = payment.booking.bookingReference;
   620→   620→
   621→   621→  // ---- Espèces : finalisation immédiate ----
   622→   622→  if (input.mode === "CASH") {
   623→   623→    const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   624→   624→      mode: "CASH",
   625→   625→      status: "PROCESSING",
   626→   626→      amount: refundAmount,
   627→   627→      initiatedByName: ctx.userName,
   628→   628→      initiatedAt: new Date().toISOString(),
   629→   629→    };
   630→   630→    await finalizeRefund(paymentId, "CASH", refundMeta, ctx.userId, ctx.ip ?? null, null);
   631→   631→    const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   632→   632→    return toPaymentDTO(fresh!);
   633→   633→  }
   634→   634→
   635→   635→  // ---- Modes MoMo : disbursement obligatoire ----
   636→   636→  if (!momoDisbursementConfigured()) {
   637→   637→    throw new ApiError(
   638→   638→      503,
   639→   639→      "MOMO_NOT_CONFIGURED",
   640→   640→      "Remboursement MTN MoMo non configuré. Renseignez les clés MOMO_DISBURSEMENT_* (voir README)."
   641→   641→    );
   642→   642→  }
   643→   643→  const referenceId = randomUUID();
   644→   644→  let msisdn: string | null = null;
   645→   645→
   646→   646→  if (input.mode === "MOMO_REFUND") {
   647→   647→    if (payment.provider !== "MTN_MOMO" || !payment.providerTransactionId || payment.providerTransactionId.startsWith("SIM-")) {
   648→   648→      throw new ApiError(
   649→   649→        409,
   650→   650→        ERROR_CODES.PAYMENT_ERROR,
   651→   651→        "Le remboursement automatique MTN nécessite un paiement MTN MoMo réel (pas une simulation)."
   652→   652→      );
   653→   653→    }
   654→   654→    await momoRefund({
   655→   655→      referenceId,
   656→   656→      amount: refundAmount,
   657→   657→      currency: momoCurrency(),
   658→   658→      externalId: bookingRef,
   659→   659→      referenceIdToRefund: payment.providerTransactionId,
   660→   660→      payerMessage: `Remboursement billet ${bookingRef}`,
   661→   661→      payeeNote: "Remboursement NZOKO TRANSPORT",
   662→   662→    });
   663→   663→  } else {
   664→   664→    // MOMO_TRANSFER — envoi de fonds vers le MSISDN du passager (ou saisi)
   665→   665→    const rawPhone: string | null = input.msisdn ?? meta.momoPhone ?? null;
   666→   666→    if (!rawPhone) throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Numéro Mobile Money du bénéficiaire requis.");
   667→   667→    msisdn = toMomoMsisdn(rawPhone);
   668→   668→    await momoTransfer({
   669→   669→      referenceId,
   670→   670→      amount: refundAmount,
   671→   671→      currency: momoCurrency(),
   672→   672→      externalId: bookingRef,
   673→   673→      msisdn,
   674→   674→      payerMessage: `Remboursement billet ${bookingRef}`,
   675→   675→      payeeNote: "Remboursement NZOKO TRANSPORT",
   676→   676→    });
   677→   677→  }
   678→   678→
   679→   679→  const refundMeta: NonNullable<PaymentMetadata["refund"]> = {
   680→   680→    mode: input.mode,
   681→   681→    status: "PROCESSING",
   682→   682→    amount: refundAmount,
   683→   683→    referenceId,
   684→   684→    msisdn,
   685→   685→    initiatedByName: ctx.userName,
   686→   686→    initiatedAt: new Date().toISOString(),
   687→   687→  };
   688→   688→  await writeMetadata(paymentId, (m) => ({ ...m, refund: refundMeta }));
   689→   689→
   690→   690→  await logAudit({
   691→   691→    userId: ctx.userId,
   692→   692→    action: "REFUND_INITIATED",
   693→   693→    entity: "Payment",
   694→   694→    entityId: paymentId,
   695→   695→    metadata: { mode: input.mode, amount: refundAmount, booking: bookingRef, momoReferenceId: referenceId },
   696→   696→    ipAddress: ctx.ip ?? null,
   697→   697→  });
   698→   698→
   699→   699→  const fresh = await db.payment.findUnique({ where: { id: paymentId }, include: { createdBy: true, booking: true } });
   700→   700→  return toPaymentDTO(fresh!);
   701→   701→}

   802→   802→// Webhook fournisseur interne — HMAC, idempotent, jamais sur parole du client
   803→   803→// POST /api/webhooks/payments
   804→   804→// Signature attendue : X-Nzoko-Signature: hex HMAC-SHA256(body, WEBHOOK_SECRET)
   805→   805→// ------------------------------------------------------------
   806→   806→export interface WebhookPayload {
   807→   807→  provider: string;
   808→   808→  providerTransactionId: string;
   809→   809→  status: "SUCCESS" | "FAILED" | "CANCELLED";
   810→   810→  amount?: number;

   813→   813→export async function handlePaymentWebhook(rawBody: string, signature: string | null, ip: string | null): Promise<{ received: true; duplicate: boolean }> {
   814→   814→  const secret = process.env.WEBHOOK_SECRET;
   815→   815→  if (!secret) throw new ApiError(503, "WEBHOOK_NOT_CONFIGURED", "Webhook non configuré.");
   816→   816→
   817→   817→  if (!signature || !timingSafeEqual(signature, hmacSha256(secret, rawBody))) {
   818→   818→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "signature invalide" } });
   819→   819→    throw new ApiError(401, ERROR_CODES.UNAUTHORIZED, "Signature webhook invalide.");
   820→   820→  }
   821→   821→
   822→   822→  const payload = safeJsonParse<WebhookPayload | null>(rawBody, null);
   823→   823→  if (!payload?.providerTransactionId || !payload.status) {
   824→   824→    throw new ApiError(400, ERROR_CODES.VALIDATION_ERROR, "Payload webhook invalide.");
   825→   825→  }
   826→   826→
   827→   827→  const payment = await db.payment.findUnique({ where: { providerTransactionId: payload.providerTransactionId } });
   828→   828→  if (!payment) {
   829→   829→    await logSecurity({ event: "WEBHOOK_REJECTED", ipAddress: ip, details: { reason: "transaction inconnue", tx: payload.providerTransactionId } });
   830→   830→    throw new ApiError(404, ERROR_CODES.NOT_FOUND, "Transaction inconnue.");
   831→   831→  }
   832→   832→
   833→   833→  if (payload.amount !== undefined && payload.amount !== payment.amount) {
   834→   834→    await logSecurity({ event: "SUSPICIOUS", ipAddress: ip, details: { reason: "montant webhook divergent", paymentId: payment.id } });
   835→   835→    throw new ApiError(409, ERROR_CODES.PAYMENT_ERROR, "Montant divergent — paiement bloqué.");
   836→   836→  }
   837→   837→
   838→   838→  if (payload.status !== "SUCCESS") {
   839→   839→    await db.payment.update({ where: { id: payment.id }, data: { status: payload.status } });
   840→   840→    return { received: true, duplicate: false };
   841→   841→  }
